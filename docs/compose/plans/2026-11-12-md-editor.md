```ts
/* Markdown 渲染：后端存 markdown 原文，前端用 marked + KaTeX 渲染并经 DOMPurify 消毒防 XSS */

import DOMPurify from 'dompurify'
import { marked } from 'marked'
import markedKatex from 'marked-katex-extension'
import 'katex/dist/katex.min.css'

marked.use(markedKatex({ throwOnError: false, nonStandard: true }))

const PURIFY_CONFIG = {
  ADD_TAGS: [
    'math', 'semantics', 'annotation', 'mrow', 'mi', 'mo', 'mn', 'msup', 'msub',
    'msubsup', 'mfrac', 'msqrt', 'mroot', 'merror', 'mtext', 'ms', 'mstyle',
    'mtable', 'mtr', 'mtd', 'munder', 'mover', 'munderover', 'mpadded',
    'mphantom', 'mspace', 'maction',
  ],
  ADD_ATTR: [
    'encoding', 'displaystyle', 'mathvariant', 'stretchy', 'fence', 'separator',
    'accent', 'accentunder', 'scriptlevel', 'display', 'linethickness', 'lspace',
    'rspace', 'voffset', 'width', 'height', 'depth',
  ],
}

export function renderMarkdown(source: string): string {
  try {
    const raw = marked.parse(source, { async: false })
    return DOMPurify.sanitize(raw, PURIFY_CONFIG)
  } catch {
    return source
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
  }
}
```

说明：`nonStandard: true` 允许 `$x$` 紧挨中文/字母（中文写作常态）；`throwOnError: false` 让错误公式显示 `katex-error` 而非抛错；DOMPurify 放行集仅覆盖 KaTeX 的 MathML 产物。

<script setup lang="ts">

import { EditorState, type Extension } from '@codemirror/state'
import { EditorView, drawSelection, keymap, placeholder as cmPlaceholder } from '@codemirror/view'
import { defaultKeymap, history, historyKeymap, indentWithTab } from '@codemirror/commands'
import { markdown } from '@codemirror/lang-markdown'

const props = withDefaults(
  defineProps<{
    modelValue: string
    placeholder?: string
    minHeight?: string
  }>(),
  { placeholder: '', minHeight: '240px' },
)

const emit = defineEmits<{ 'update:modelValue': [value: string] }>()

const host = ref<HTMLElement | null>(null)
let view: EditorView | null = null

const theme = EditorView.theme({
  '&': {
    fontSize: '0.95rem',
    color: 'var(--ink)',
    backgroundColor: 'var(--bg-raised)',
    border: 'var(--bw-sm) solid var(--border)',
    borderRadius: '8px',
  },
  '&.cm-focused': { outline: 'none', borderColor: 'var(--accent-interactive)' },
  '.cm-content': {
    fontFamily: 'ui-monospace, SFMono-Regular, Menlo, Consolas, monospace',
    padding: '10px 12px',
  },
  '.cm-scroller': { maxHeight: '32rem' },
  '.cm-cursor': { borderLeftColor: 'var(--ink)' },
  '.cm-selectionBackground': { backgroundColor: 'var(--clay-soft)' },
})

onMounted(() => {
  if (!host.value) return
  const extensions: Extension[] = [
    markdown(),
    history(),
    drawSelection(),
    EditorView.lineWrapping,
    theme,
    cmPlaceholder(props.placeholder),
    keymap.of([...defaultKeymap, ...historyKeymap, indentWithTab]),
    EditorView.updateListener.of((update) => {
      if (update.docChanged) {
        emit('update:modelValue', update.state.doc.toString())
      }
    }),
  ]
  view = new EditorView({
    state: EditorState.create({ doc: props.modelValue, extensions }),
    parent: host.value,
  })
})

watch(
  () => props.modelValue,
  (value) => {
    if (view && value !== view.state.doc.toString()) {
      view.dispatch({ changes: { from: 0, to: view.state.doc.length, insert: value } })
    }
  },
)

onBeforeUnmount(() => {
  view?.destroy()
  view = null
})
</script>

<template>
  <div class="md-editor">
    <div ref="host" class="md-pane md-edit" />
  </div>
</template>

<style scoped>
.md-editor {
  --md-min-height: v-bind(minHeight);
}

.md-pane {
  min-height: var(--md-min-height);
}

.md-edit :deep(.cm-editor) {
  min-height: var(--md-min-height);
}
</style>

function wrapSelection(before: string, after: string): void {
  if (!view) return
  const { state } = view
  const changes = state.changeByRange((range) => ({
    changes: [
      {
        from: range.from,
        to: range.to,
        insert: before + state.sliceDoc(range.from, range.to) + after,
      },
    ],
    range: EditorSelection.range(range.from + before.length, range.to + before.length),
  }))
  view.dispatch(changes, { scrollIntoView: true })
}

function prefixLines(prefix: string): void {
  if (!view) return
  const { state } = view
  const changes = state.changeByRange((range) => {
    const first = state.doc.lineAt(range.from).number
    const last = state.doc.lineAt(range.to).number
    const lineChanges = []
    for (let n = first; n <= last; n += 1) {
      lineChanges.push({ from: state.doc.line(n).from, insert: prefix })
    }
    return { changes: lineChanges }
  })
  view.dispatch(changes, { scrollIntoView: true })
}

function insertBlock(text: string, caretOffset = text.length): void {
  if (!view) return
  const range = view.state.selection.main
  view.dispatch({
    changes: { from: range.from, to: range.to, insert: text },
    selection: EditorSelection.cursor(range.from + caretOffset),
    scrollIntoView: true,
  })
}

type Mode = 'edit' | 'preview' | 'split'
const mode = ref<Mode>('split')
const syncScroll = ref(true)
const previewEl = ref<HTMLElement | null>(null)

const previewHtml = computed(() => renderMarkdown(props.modelValue))

function syncFrom(src: HTMLElement, dest: HTMLElement): void {
  if (syncing || !syncScroll.value || mode.value !== 'split') return
  syncing = true
  const max = src.scrollHeight - src.clientHeight
  const ratio = max > 0 ? src.scrollTop / max : 0
  dest.scrollTop = ratio * (dest.scrollHeight - dest.clientHeight)
  requestAnimationFrame(() => {
    syncing = false
  })
}

function onEditorScroll(): void {
  if (view && previewEl.value) syncFrom(view.scrollDOM, previewEl.value)
}

function onPreviewScroll(): void {
  if (view && previewEl.value) syncFrom(previewEl.value, view.scrollDOM)
}
